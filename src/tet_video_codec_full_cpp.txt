// tet_video_codec_full.cpp
//
// Whole-video 3D-tet modal codec:
// - Tiles the video into bricks Bx x By x Bt in (x,y,t)
// - Each brick is [0,1]^3 in local coords, split into 6 tets
// - Each tet gets modal Jacobi coefficients (3D simplex, degree p)
// - Encode: read full color video, write raw coeffs to text
// - Decode: read coeffs, reconstruct full-color video, write recon.mp4
//
// This version:
// - Counts the actually decodable frames first (no trust in CAP_PROP_FRAME_COUNT)
// - Encodes only N_enc = floor(N_real / Bt) * Bt frames
// - Drops the last partial slab if needed, but never errors out on it
// - Uses bilinear sampling in x,y (nearest in t) on encode
// - Uses robust per-pixel, per-tet evaluation on decode (no holes)
// - Supports temporal supersampling on decode by factor upsampleT
//
// Build (Ubuntu, with libopencv-dev installed):
//   g++ tet_video_codec_full.cpp -o tet_video_codec_full `pkg-config --cflags --libs opencv4`
//
// Make sure jbasis.hh and the tet quadrature file are in your include / runtime paths.

#include <vector>
#include <array>
#include <fstream>
#include <iostream>
#include <cassert>
#include <cmath>
#include <algorithm>
#include <string>

#include <opencv2/opencv.hpp>

// Adjust this include path to your actual location
#include "jbasis.hh"

using jsimplex::Basis;

// ---------------- Basic structs ----------------

struct Vec3 {
  double x, y, z;
};

struct Tet {
  int v[4];  // indices into the brick-vertex array
};

// Unit cube vertices in (x,y,t) ∈ [0,1]^3
static const std::array<Vec3, 8> cubeVerts = {{
  {0.0, 0.0, 0.0}, // v0
  {1.0, 0.0, 0.0}, // v1
  {1.0, 1.0, 0.0}, // v2
  {0.0, 1.0, 0.0}, // v3
  {0.0, 0.0, 1.0}, // v4
  {1.0, 0.0, 1.0}, // v5
  {1.0, 1.0, 1.0}, // v6
  {0.0, 1.0, 1.0}  // v7
}};

// 6-tet decomposition of the cube using body diagonal v0->v6
static const std::array<Tet, 6> cubeTets = {{
  {{0, 1, 2, 6}},
  {{0, 2, 3, 6}},
  {{0, 3, 7, 6}},
  {{0, 7, 4, 6}},
  {{0, 4, 5, 6}},
  {{0, 5, 1, 6}}
}};

// ---------------- Vector ops & affine maps ----------------

inline Vec3 operator+(const Vec3 &a, const Vec3 &b) {
  return {a.x + b.x, a.y + b.y, a.z + b.z};
}
inline Vec3 operator-(const Vec3 &a, const Vec3 &b) {
  return {a.x - b.x, a.y - b.y, a.z - b.z};
}
inline Vec3 operator*(double s, const Vec3 &a) {
  return {s * a.x, s * a.y, s * a.z};
}

// Map reference tet point r=(r,s,t) to physical cube tet K in [0,1]^3
Vec3 map_ref_to_tet(const Vec3 &r,
                    const std::array<Vec3,8> &verts,
                    const Tet &K) {
  const Vec3 &v0 = verts[K.v[0]];
  const Vec3 &v1 = verts[K.v[1]];
  const Vec3 &v2 = verts[K.v[2]];
  const Vec3 &v3 = verts[K.v[3]];

  Vec3 e1 = v1 - v0;
  Vec3 e2 = v2 - v0;
  Vec3 e3 = v3 - v0;

  return v0 + r.x * e1 + r.y * e2 + r.z * e3;
}

// |det J_K| for tet K in the cube
double tet_jacobian_det(const std::array<Vec3,8> &verts,
                        const Tet &K) {
  const Vec3 &v0 = verts[K.v[0]];
  const Vec3 &v1 = verts[K.v[1]];
  const Vec3 &v2 = verts[K.v[2]];
  const Vec3 &v3 = verts[K.v[3]];

  Vec3 e1 = v1 - v0;
  Vec3 e2 = v2 - v0;
  Vec3 e3 = v3 - v0;

  double det =
      e1.x * (e2.y * e3.z - e2.z * e3.y)
    - e1.y * (e2.x * e3.z - e2.z * e3.x)
    + e1.z * (e2.x * e3.y - e2.y * e3.x);

  return std::fabs(det);
}

// Compute inverse of B=[e1 e2 e3] (columns), store in B_inv as row-major 3x3
void compute_B_inv(const Vec3 &e1, const Vec3 &e2, const Vec3 &e3,
                   double B_inv[9]) {
  // Column vectors e1, e2, e3 -> matrix B
  double b00 = e1.x, b10 = e1.y, b20 = e1.z;
  double b01 = e2.x, b11 = e2.y, b21 = e2.z;
  double b02 = e3.x, b12 = e3.y, b22 = e3.z;

  double det =
      b00 * (b11 * b22 - b12 * b21)
    - b01 * (b10 * b22 - b12 * b20)
    + b02 * (b10 * b21 - b11 * b20);

  double invDet = 1.0 / det;

  // Row 0 of B^{-1}
  B_inv[0] = (b11 * b22 - b12 * b21) * invDet;
  B_inv[1] = (b02 * b21 - b01 * b22) * invDet;
  B_inv[2] = (b01 * b12 - b02 * b11) * invDet;

  // Row 1
  B_inv[3] = (b12 * b20 - b10 * b22) * invDet;
  B_inv[4] = (b00 * b22 - b02 * b20) * invDet;
  B_inv[5] = (b02 * b10 - b00 * b12) * invDet;

  // Row 2
  B_inv[6] = (b10 * b21 - b11 * b20) * invDet;
  B_inv[7] = (b01 * b20 - b00 * b21) * invDet;
  B_inv[8] = (b00 * b11 - b01 * b10) * invDet;
}

// ---------------- Quadrature loader ----------------

bool load_tet_quadrature(const std::string &filename,
                         std::vector<Vec3> &Xref,
                         std::vector<double> &W) {
  std::ifstream in(filename);
  if (!in) {
    std::cerr << "Could not open quadrature file: " << filename << "\n";
    return false;
  }

  std::vector<double> data;
  double val;
  while (in >> val) {
    data.push_back(val);
  }

  const int D = 3;
  if (data.size() % (D + 1) != 0) {
    std::cerr << "Quadrature data length not divisible by D+1.\n";
    return false;
  }

  int N = static_cast<int>(data.size()) / (D + 1);
  Xref.resize(N);
  W.resize(N);

  for (int q = 0; q < N; ++q) {
    Xref[q].x = data[q * D + 0];
    Xref[q].y = data[q * D + 1];
    Xref[q].z = data[q * D + 2];
  }
  for (int q = 0; q < N; ++q) {
    W[q] = data[N * D + q];
  }

  return true;
}

// ---------------- Basis wrapper ----------------

struct TetBasis3D {
  int p;         // total degree
  int M;         // dim_Pi(p)
  std::vector<int> alpha_table;
  std::vector<int> tail_deg;
  std::vector<double> inv_h;

  // Optionally cached values at reference quadrature points
  std::vector<double> Vref;
  int ld_V;      // leading dimension (Nq)

  void init_with_quadrature(int degree,
                            const double *kappa,
                            const std::vector<Vec3> &Xref) {
    p = degree;
    M = Basis<3,double>::dim_Pi(p);

    alpha_table.resize(M * 3);
    tail_deg.resize(M * 3);
    inv_h.resize(M);

    Basis<3,double>::build_structures(
      kappa, p,
      alpha_table.data(),
      tail_deg.data(),
      inv_h.data()
    );

    int Nq = static_cast<int>(Xref.size());
    ld_V = Nq;
    Vref.assign(M * ld_V, 0.0);

    std::vector<double> Xraw(Nq * 3);
    for (int q = 0; q < Nq; ++q) {
      Xraw[q * 3 + 0] = Xref[q].x;
      Xraw[q * 3 + 1] = Xref[q].y;
      Xraw[q * 3 + 2] = Xref[q].z;
    }

    Basis<3,double>::eval_all(
      Xraw.data(),
      3,
      1,
      Nq,
      kappa,
      p,
      alpha_table.data(),
      tail_deg.data(),
      inv_h.data(),
      Vref.data(),
      ld_V,
      nullptr
    );
  }

  void init_structures_only(int degree, const double *kappa) {
    p = degree;
    M = Basis<3,double>::dim_Pi(p);
    alpha_table.resize(M * 3);
    tail_deg.resize(M * 3);
    inv_h.resize(M);
    Basis<3,double>::build_structures(
      kappa, p,
      alpha_table.data(),
      tail_deg.data(),
      inv_h.data()
    );
    ld_V = 0;
    Vref.clear();
  }

  inline double phi_ref(int m, int q) const {
    return Vref[q + m * ld_V];
  }
};

// ---------------- Video brick sampler (color, bilinear in x,y) ----------------

struct VideoBrickSampler {
  int imgW, imgH;
  int Bx, By, Bt;
  int bx0, by0;
  int t0;
  const std::vector<cv::Mat> *frames;  // pointer to slab frames (CV_8UC3)

  // uvw ∈ [0,1]^3 -> sample channel c ∈ {0,1,2}
  // Bilinear in x,y (within the brick), nearest neighbour in t.
  double sample_local(const Vec3 &uvw, int c) const {
    auto clamp01m = [](double s) {
      return std::min(std::max(s, 0.0), std::nextafter(1.0, 0.0));
    };

    double u = clamp01m(uvw.x);
    double v = clamp01m(uvw.y);
    double w = clamp01m(uvw.z);

    // ---- time: nearest frame in the brick ----
    int lt;
    if (Bt <= 1) {
      lt = 0;
    } else {
      lt = static_cast<int>(std::round(w * (Bt - 1)));
      lt = std::min(std::max(lt, 0), Bt - 1);
    }

    // ---- space: bilinear within the brick ----
    double x_f = (Bx > 1) ? u * (Bx - 1) : 0.0;
    double y_f = (By > 1) ? v * (By - 1) : 0.0;

    int ix0 = static_cast<int>(std::floor(x_f));
    int iy0 = static_cast<int>(std::floor(y_f));
    int ix1 = ix0 + 1;
    int iy1 = iy0 + 1;

    double ax = x_f - ix0;
    double ay = y_f - iy0;

    ix0 = std::min(std::max(ix0, 0), Bx - 1);
    iy0 = std::min(std::max(iy0, 0), By - 1);
    ix1 = std::min(std::max(ix1, 0), Bx - 1);
    iy1 = std::min(std::max(iy1, 0), By - 1);

    int x0 = bx0 + ix0;
    int y0 = by0 + iy0;
    int x1 = bx0 + ix1;
    int y1 = by0 + iy1;

    x0 = std::min(std::max(x0, 0), imgW - 1);
    y0 = std::min(std::max(y0, 0), imgH - 1);
    x1 = std::min(std::max(x1, 0), imgW - 1);
    y1 = std::min(std::max(y1, 0), imgH - 1);

    const cv::Mat &frame = frames->at(lt);

    const cv::Vec3b &p00 = frame.at<cv::Vec3b>(y0, x0);
    const cv::Vec3b &p10 = frame.at<cv::Vec3b>(y0, x1);
    const cv::Vec3b &p01 = frame.at<cv::Vec3b>(y1, x0);
    const cv::Vec3b &p11 = frame.at<cv::Vec3b>(y1, x1);

    double v00 = static_cast<double>(p00[c]);
    double v10 = static_cast<double>(p10[c]);
    double v01 = static_cast<double>(p01[c]);
    double v11 = static_cast<double>(p11[c]);

    double wx0 = 1.0 - ax;
    double wy0 = 1.0 - ay;

    double value =
      wx0 * wy0 * v00 +
      ax  * wy0 * v10 +
      wx0 * ay  * v01 +
      ax  * ay  * v11;

    return value;
  }
};

// ---------------- Coefficient file format ----------------
//
// Global header (single line):
// W H Nf_enc Bx By Bt degree numTets M numChannels
//
// Nf_enc = number of frames actually encoded (multiple of Bt).
//
// Then for each brick in encode order:
//   brick header line:
//     t0 bx0 by0
//   then for each channel c=0..numChannels-1:
//     for each tet k=0..numTets-1:
//       one line with M doubles: c_0 ... c_{M-1}
//
// numTets=6, numChannels=3 (B,G,R).
//

// Count decodable frames and get frame size by actually reading them
int count_decodable_frames_and_size(const std::string &videoFile,
                                    int &W, int &H) {
  cv::VideoCapture cap(videoFile);
  if (!cap.isOpened()) {
    std::cerr << "Could not open video for counting: " << videoFile << "\n";
    W = H = -1;
    return -1;
  }

  int count = 0;
  cv::Mat frame;
  W = H = -1;

  while (true) {
    if (!cap.read(frame)) break;
    if (count == 0) {
      W = frame.cols;
      H = frame.rows;
    }
    ++count;
  }

  return count;
}

// Encode full video
bool encode_video_full(const std::string &videoFile,
                       const std::string &quadFile,
                       const std::string &coeffFile,
                       int Bx, int By, int Bt,
                       int degree) {
  int W = -1, H = -1;
  int Nf_real = count_decodable_frames_and_size(videoFile, W, H);
  if (Nf_real <= 0) {
    std::cerr << "No decodable frames in video.\n";
    return false;
  }

  int Nf_enc = (Nf_real / Bt) * Bt;
  if (Nf_enc == 0) {
    std::cerr << "Not enough frames for a single full slab (Bt=" << Bt << ").\n";
    return false;
  }

  if (W % Bx != 0 || H % By != 0) {
    std::cerr << "For now, require W % Bx == 0 and H % By == 0.\n";
    std::cerr << "Got W=" << W << " H=" << H
              << " Bx=" << Bx << " By=" << By << "\n";
    return false;
  }

  int numTets = static_cast<int>(cubeTets.size());
  int numChannels = 3;

  std::vector<Vec3> Xref;
  std::vector<double> Wq;
  if (!load_tet_quadrature(quadFile, Xref, Wq)) {
    return false;
  }
  int Nq = static_cast<int>(Xref.size());

  double kappa[4] = {0.5, 0.5, 0.5, 0.5};
  TetBasis3D basis;
  basis.init_with_quadrature(degree, kappa, Xref);
  int M = basis.M;

  std::cout << "Encoding full video:\n";
  std::cout << "  Video: " << videoFile << " ("
            << W << "x" << H << ", decodable frames = " << Nf_real << ")\n";
  std::cout << "  Encoding first Nf_enc = " << Nf_enc << " frames"
            << " (dropping last " << (Nf_real - Nf_enc) << " if any)\n";
  std::cout << "  Brick: " << Bx << "x" << By << "x" << Bt
            << ", degree=" << degree
            << ", M=" << M
            << ", numTets=" << numTets
            << "\n";

  std::ofstream out(coeffFile);
  if (!out) {
    std::cerr << "Could not open coeff output file: " << coeffFile << "\n";
    return false;
  }

  out << W << " " << H << " " << Nf_enc << " "
      << Bx << " " << By << " " << Bt << " "
      << degree << " " << numTets << " " << M << " "
      << numChannels << "\n";

  cv::VideoCapture cap(videoFile);
  if (!cap.isOpened()) {
    std::cerr << "Could not open video for encoding: " << videoFile << "\n";
    return false;
  }

  std::vector<cv::Mat> frames_slab;
  int t0 = 0;

  while (t0 < Nf_enc) {
    frames_slab.clear();
    frames_slab.reserve(Bt);

    for (int k = 0; k < Bt; ++k) {
      cv::Mat frameBGR;
      if (!cap.read(frameBGR)) {
        std::cerr << "Unexpected short read during encoding at t0=" << t0
                  << ", slab frame " << k << "\n";
        break;
      }
      frames_slab.push_back(frameBGR);
    }

    if ((int)frames_slab.size() < Bt) {
      std::cerr << "Stopping early: only " << frames_slab.size()
                << " frames read in last slab at t0=" << t0 << "\n";
      break;
    }

    for (int by0 = 0; by0 < H; by0 += By) {
      for (int bx0 = 0; bx0 < W; bx0 += Bx) {
        VideoBrickSampler sampler;
        sampler.imgW = W;
        sampler.imgH = H;
        sampler.Bx = Bx;
        sampler.By = By;
        sampler.Bt = Bt;
        sampler.bx0 = bx0;
        sampler.by0 = by0;
        sampler.t0 = t0;
        sampler.frames = &frames_slab;

        out << t0 << " " << bx0 << " " << by0 << "\n";

        std::vector<std::vector<double>> coeffs(numChannels);
        for (int c = 0; c < numChannels; ++c) {
          coeffs[c].assign(numTets * M, 0.0);
        }

        for (int kt = 0; kt < numTets; ++kt) {
          const Tet &K = cubeTets[kt];

          for (int q = 0; q < Nq; ++q) {
            Vec3 r = Xref[q];
            Vec3 uvw = map_ref_to_tet(r, cubeVerts, K);
            double w_q = Wq[q];

            double u_c[3];
            for (int c = 0; c < numChannels; ++c) {
              u_c[c] = sampler.sample_local(uvw, c);
            }

            for (int m = 0; m < M; ++m) {
              double phi = basis.phi_ref(m, q);
              for (int c = 0; c < numChannels; ++c) {
                coeffs[c][kt * M + m] += w_q * u_c[c] * phi;
              }
            }
          }
        }

        for (int c = 0; c < numChannels; ++c) {
          for (int kt = 0; kt < numTets; ++kt) {
            for (int m = 0; m < M; ++m) {
              out << coeffs[c][kt * M + m];
              if (m + 1 < M) out << " ";
            }
            out << "\n";
          }
        }
      }
    }

    std::cout << "Encoded slab t0=" << t0 << " ... t0+Bt=" << (t0 + Bt) << "\n";
    t0 += Bt;
  }

  std::cout << "Wrote coefficients to " << coeffFile << "\n";
  return true;
}

// ---------------- Decode full video from coeffs (with time supersampling) -----

struct CoeffHeader {
  int W, H, Nf;
  int Bx, By, Bt;
  int degree;
  int numTets;
  int M;
  int numChannels;
};

bool read_global_header(std::istream &in, CoeffHeader &hdr) {
  in >> hdr.W >> hdr.H >> hdr.Nf
     >> hdr.Bx >> hdr.By >> hdr.Bt
     >> hdr.degree >> hdr.numTets >> hdr.M
     >> hdr.numChannels;
  return (bool)in;
}

// upsampleT >= 1: factor by which to upsample time on decode.
// Nf_out = Nf * upsampleT.
bool decode_video_full(const std::string &coeffFile,
                       std::vector<cv::Mat> &framesOut,
                       int upsampleT) {
  std::ifstream in(coeffFile);
  if (!in) {
    std::cerr << "Could not open coeff file: " << coeffFile << "\n";
    return false;
  }

  if (upsampleT < 1) upsampleT = 1;

  CoeffHeader hdr;
  if (!read_global_header(in, hdr)) {
    std::cerr << "Failed to read global header from coeff file.\n";
    return false;
  }

  if (hdr.numChannels != 3) {
    std::cerr << "Expected numChannels=3, got " << hdr.numChannels << "\n";
    return false;
  }

  int W  = hdr.W;
  int H  = hdr.H;
  int Nf = hdr.Nf;     // encoded frame count
  int Bx = hdr.Bx;
  int By = hdr.By;
  int Bt = hdr.Bt;
  int degree    = hdr.degree;
  int numTets   = hdr.numTets;
  int M         = hdr.M;

  int Nf_out = Nf * upsampleT;

  // Allocate output frames
  framesOut.clear();
  framesOut.resize(Nf_out);
  for (int t = 0; t < Nf_out; ++t) {
    framesOut[t] = cv::Mat(H, W, CV_8UC3, cv::Scalar(0, 0, 0));
  }

  // Basis structures
  double kappa[4] = {0.5, 0.5, 0.5, 0.5};
  TetBasis3D basis;
  basis.init_structures_only(degree, kappa);

  // Precompute tet affine inverses on the unit cube
  struct TetAffine {
    Vec3 v0;
    double B_inv[9];
  };
  std::vector<TetAffine> taff(numTets);

  for (int kt = 0; kt < numTets; ++kt) {
    const Tet &K = cubeTets[kt];
    Vec3 v0 = cubeVerts[K.v[0]];
    Vec3 v1 = cubeVerts[K.v[1]];
    Vec3 v2 = cubeVerts[K.v[2]];
    Vec3 v3 = cubeVerts[K.v[3]];

    Vec3 e1 = v1 - v0;
    Vec3 e2 = v2 - v0;
    Vec3 e3 = v3 - v0;

    taff[kt].v0 = v0;
    compute_B_inv(e1, e2, e3, taff[kt].B_inv);
  }

  const double eps = 1e-12;

  // Brick iteration order must match encoder's
  for (int t0 = 0; t0 < Nf; t0 += Bt) {
    for (int by0 = 0; by0 < H; by0 += By) {
      for (int bx0 = 0; bx0 < W; bx0 += Bx) {
        // --- read brick header ---
        int t0_f, bx0_f, by0_f;
        in >> t0_f >> bx0_f >> by0_f;
        if (!in) {
          std::cerr << "Error reading brick header.\n";
          return false;
        }
        if (t0_f != t0 || bx0_f != bx0 || by0_f != by0) {
          std::cerr << "Brick header mismatch at t0=" << t0
                    << " bx0=" << bx0 << " by0=" << by0 << "\n";
          return false;
        }

        // --- read coeffs for this brick: [3][numTets*M] ---
        std::vector<std::vector<double>> coeffs(3);
        for (int c = 0; c < 3; ++c) {
          coeffs[c].assign(numTets * M, 0.0);
          for (int kt = 0; kt < numTets; ++kt) {
            for (int m = 0; m < M; ++m) {
              in >> coeffs[c][kt * M + m];
              if (!in) {
                std::cerr << "Error reading coeffs.\n";
                return false;
              }
            }
          }
        }

        int Nx = Bx;
        int Ny = By;
        int Nt = Bt * upsampleT;  // supersampled time samples per brick
        int totalSamples = Nx * Ny * Nt;

        // uhat[c][sample index]
        std::vector<double> uhat[3];
        for (int c = 0; c < 3; ++c) {
          uhat[c].assign(totalSamples, 0.0);
        }

        struct SampleRef {
          int idx;
          Vec3 r;
        };

        // sample lists per tet
        std::vector<std::vector<SampleRef>> samplesPerTet(numTets);

        // --- classify samples into tets ---
        int idx = 0;
        for (int it = 0; it < Nt; ++it) {
          double w = (it + 0.5) / Nt;  // finer grid in [0,1]
          for (int iy = 0; iy < Ny; ++iy) {
            double v = (iy + 0.5) / Ny;
            for (int ix = 0; ix < Nx; ++ix) {
              double u = (ix + 0.5) / Nx;
              Vec3 x = {u, v, w};

              bool assigned = false;
              for (int kt = 0; kt < numTets && !assigned; ++kt) {
                const TetAffine &TA = taff[kt];
                Vec3 dx = x - TA.v0;
                const double *B = TA.B_inv;
                Vec3 r;
                r.x = B[0]*dx.x + B[1]*dx.y + B[2]*dx.z;
                r.y = B[3]*dx.x + B[4]*dx.y + B[5]*dx.z;
                r.z = B[6]*dx.x + B[7]*dx.y + B[8]*dx.z;

                double s = r.x + r.y + r.z;
                if (r.x >= -eps && r.y >= -eps && r.z >= -eps && s <= 1.0 + eps) {
                  samplesPerTet[kt].push_back({idx, r});
                  assigned = true;
                }
              }
              // if not assigned, uhat stays 0 (should not happen)
              ++idx;
            }
          }
        }

        // --- evaluate basis and reconstruct per tet ---
        for (int kt = 0; kt < numTets; ++kt) {
          auto &lst = samplesPerTet[kt];
          int npts = (int)lst.size();
          if (npts == 0) continue;

          std::vector<double> Xraw(npts * 3);
          for (int i = 0; i < npts; ++i) {
            Xraw[3*i + 0] = lst[i].r.x;
            Xraw[3*i + 1] = lst[i].r.y;
            Xraw[3*i + 2] = lst[i].r.z;
          }

          std::vector<double> V_eval(M * npts);
          int ld_V = npts;

          Basis<3,double>::eval_all(
            Xraw.data(),
            3,
            1,
            npts,
            kappa,
            degree,
            basis.alpha_table.data(),
            basis.tail_deg.data(),
            basis.inv_h.data(),
            V_eval.data(),
            ld_V,
            nullptr
          );

          for (int i = 0; i < npts; ++i) {
            int sidx = lst[i].idx;
            for (int c = 0; c < 3; ++c) {
              const auto &ck = coeffs[c];
              double val = 0.0;
              for (int m = 0; m < M; ++m) {
                val += ck[kt * M + m] * V_eval[i + m * ld_V];
              }
              uhat[c][sidx] = val;
            }
          }
        }

        // --- paste brick into global frames ---
        idx = 0;
        for (int it = 0; it < Nt; ++it) {
          int tGlobal = t0 * upsampleT + it;  // upsampled global time index
          if (tGlobal >= Nf_out) continue;
          for (int iy = 0; iy < Ny; ++iy) {
            int yGlobal = by0 + iy;
            if (yGlobal >= H) continue;
            cv::Vec3b *row = framesOut[tGlobal].ptr<cv::Vec3b>(yGlobal);
            for (int ix = 0; ix < Nx; ++ix) {
              int xGlobal = bx0 + ix;
              if (xGlobal >= W) continue;
              double b = uhat[0][idx];
              double g = uhat[1][idx];
              double r = uhat[2][idx];
              b = std::clamp(b, 0.0, 255.0);
              g = std::clamp(g, 0.0, 255.0);
              r = std::clamp(r, 0.0, 255.0);
              row[xGlobal] = cv::Vec3b(
                (uchar)std::round(b),
                (uchar)std::round(g),
                (uchar)std::round(r)
              );
              ++idx;
            }
          }
        }
      }
    }

    std::cout << "Decoded slab t0=" << t0
              << " ... t0+Bt=" << (t0 + Bt)
              << " (upsampleT=" << upsampleT << ")\n";
  }

  std::cout << "Decoded full video from " << coeffFile
            << " with Nf_out=" << Nf_out << " frames\n";
  return true;
}

// ---------------- Write video helper ----------------

bool write_video_color(const std::string &outFile,
                       const std::vector<cv::Mat> &frames,
                       double fps) {
  if (frames.empty()) {
    std::cerr << "No frames to write.\n";
    return false;
  }
  int W = frames[0].cols;
  int H = frames[0].rows;

  // Quantize fps to something MPEG-4 is happy with
  double fps_q = std::round(fps);
  if (fps_q < 1.0)   fps_q = 1.0;
  if (fps_q > 240.0) fps_q = 240.0;  // arbitrary upper clamp

  int fourcc = cv::VideoWriter::fourcc('m','p','4','v');
  cv::VideoWriter writer(outFile, fourcc, fps_q, cv::Size(W, H), true);
  if (!writer.isOpened()) {
    std::cerr << "Could not open VideoWriter for " << outFile << "\n";
    return false;
  }

  for (const auto &f : frames) {
    writer.write(f);
  }

  std::cout << "Wrote video to " << outFile
            << " at fps=" << fps_q << " (requested " << fps << ")\n";
  return true;
}

// ---------------- main (demo) ----------------

int main(int argc, char* argv[]) {
  if (argc < 2) {
    std::cerr << "Usage: " << argv[0] << " input.mp4\n";
    return 1;
  }

  std::string quadFile  = "t3_kappa0.5-0.5-0.5-0.5_N364_M1330_n11_m18.txt";
  std::string videoFile = argv[1];
  std::string coeffFile = "coeffs_full.txt";
  std::string reconFile = "recon.mp4";

  int Bx = 10;
  int By = 10;
  int Bt = 10;
  int degree = 9;

  // Choose temporal supersampling factor
  int upsampleT = 10;  // e.g. 1 = no upsample, 2 = 2x frames, 4 = 4x, ...

  cv::VideoCapture cap(videoFile);
  if (!cap.isOpened()) {
    std::cerr << "Could not open video for fps: " << videoFile << "\n";
    return 1;
  }
  double fps = cap.get(cv::CAP_PROP_FPS);
  if (fps <= 0.0) fps = 30.0; // fallback

  if (!encode_video_full(videoFile, quadFile, coeffFile,
                         Bx, By, Bt, degree)) {
    std::cerr << "Encode failed.\n";
    return 1;
  }

  std::vector<cv::Mat> reconFrames;
  if (!decode_video_full(coeffFile, reconFrames, upsampleT)) {
    std::cerr << "Decode failed.\n";
    return 1;
  }

  // Multiply fps by upsampleT so playback speed matches original
  if (!write_video_color(reconFile, reconFrames, fps)) {
    std::cerr << "Write recon video failed.\n";
    return 1;
  }

  std::cout << "Done.\n";
  return 0;
}
