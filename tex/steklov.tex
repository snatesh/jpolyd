\begin{abstract}
We formulate the hierarchical Poincar\'e--Steklov (HPS) method from the
continuous inhomogeneous Dirichlet problem through local Dirichlet-to-Neumann
(DtN) decomposition, a promoted Jacobi strong-residual Petrov--Galerkin
leaf construction, affine moment-space Steklov maps, two-subdomain merging,
root boundary solves, and downward volume reconstruction.  The leaf retains
the existing expanded strong-form differential operator and stacked QR
least-squares extension.  Its augmented outward co-normal flux is the
physical co-normal moment minus the positively weighted local
trace--skeleton mismatch.  The augmentation is the exact skeleton
gradient of the reduced local least-squares energy and defines a symmetric
positive semidefinite operator.  We distinguish this algebraic fact from a
still-conditional global coercivity statement in the presence of advection
and reaction.  Source-dependent affine terms and an explicit reusable
upward source pass are included.  The later treatment of arbitrary-
dimensional Jacobi recurrences, natural stencils, differentiation,
parameter promotion, and multiplication is retained from the prior note.
\end{abstract}

\section{Continuous divergence-form Dirichlet and co-normal problem}

Let $\Omega\subset\mathbb R^D$ be a Lipschitz domain partitioned into
nondegenerate affine $D$-simplices and, recursively, unions of simplices.
On a subdomain $K\subset\Omega$, consider
\begin{equation}\label{eq:full_cont_operator}
 \mathcal L_Ku
 =-\nabla\cdot(A_K\nabla u)+\bm b_K\cdot\nabla u+c_Ku=f_K
 \quad\text{in }K,
 \qquad u=h_K\quad\text{on }\partial K.
\end{equation}
The diffusion tensor $A=A^\top$ is uniformly positive definite within
each coefficient region, and $A,\bm b,c$ have enough elementwise
regularity for the traces below.  Our chosen physical outward co-normal
derivative is
\begin{equation}\label{eq:full_cont_q}
 q_K(u)=\bm n_K^\top A_K\nabla u\quad\text{on }\partial K.
\end{equation}
Piecewise-discontinuous coefficients are permitted, with transmission
conditions
\begin{equation}\label{eq:full_transmission}
 [u]=0\quad\text{and}\quad q_A(u_A)+q_B(u_B)=0
\end{equation}
for opposing outward normals in the absence of singular interface sources.  This is the divergence-form
\emph{co-normal} problem.  If $\bm b\cdot\bm n$ is also continuous across
an interface, the advective contributions cancel in the total flux
$\bm b u-A\nabla u$ as well.  For a discontinuous normal advective field,
the co-normal and total-transport transmission models should not be
conflated.

Assume the Dirichlet realization of $\mathcal L_K$ is uniquely solvable
for the data under consideration.  For example, a standard sufficient
condition for the corresponding energy argument is uniform ellipticity,
suitable coefficient regularity, and
$c_{\mathrm{eff}}:=c-\tfrac12\nabla\cdot\bm b\ge0$; other well-posed
coefficient regimes are possible.  Denote the zero-boundary solution
operator by $G_K$ and the homogeneous interior Dirichlet extension by
$H_K$:
\begin{align}\label{eq:full_cont_splitting}
 \mathcal L_K(H_Kh)&=0, & H_Kh|_{\partial K}&=h,\\
 \mathcal L_K(G_Kf)&=f, & G_Kf|_{\partial K}&=0.
\end{align}
Linearity and uniqueness give the exact decomposition
\begin{equation}\label{eq:full_cont_u_split}
 u=H_Kh+G_Kf.
\end{equation}
Applying the outward co-normal trace to each component defines the
\emph{continuous Dirichlet-to-Neumann operator} $N_K$ and the
source-induced co-normal flux $q_K^f$:
\begin{equation}\label{eq:full_cont_dtn}
 N_Kh:=q_K(H_Kh),\qquad
 q_K^f:=q_K(G_Kf),\qquad
 \boxed{q_K(u)=N_Kh+q_K^f.}
\end{equation}
In standard Sobolev terminology $N_K:H^{1/2}(\partial K)\to
H^{-1/2}(\partial K)$, interpreted appropriately for the stated
coefficient assumptions.  The source-induced term is affine data, not
part of $N_K$.  If $\bm b\ne0$, $N_K$ is generally nonsymmetric.

\paragraph{What HPS propagates.}
For each subdomain $K$, hierarchical Poincar\'e--Steklov reduction
propagates the pair $(N_K,q_K^f)$: a homogeneous boundary response and a
particular source response.  Two child subdomains are joined by matching
the Dirichlet trace and balancing their opposing outward co-normal fluxes.
This is the continuum motivation for the discrete maps derived below.

\subsection{Continuous two-subdomain transmission}
Suppose $K=K_A\cup K_B$, with shared interface
$I=\partial K_A\cap\partial K_B$ and external portions $E_A,E_B$.
Prescribe $h_A$ on $E_A$ and $h_B$ on $E_B$, and introduce the unknown
common interface Dirichlet value $\gamma$:
\begin{equation}\label{eq:full_cont_children_dirichlet}
 h_{K_A}=(h_A,\gamma),\qquad
 h_{K_B}=(h_B,\gamma).
\end{equation}
For continuous co-normal traces, the interface equation is
\begin{equation}\label{eq:full_cont_interface}
 \left[N_A(h_A,\gamma)+q_A^f\right]_{I}
 +\left[N_B(h_B,\gamma)+q_B^f\right]_{I}=0.
\end{equation}
Once solved for $\gamma$, the external components of the two child
fluxes define $(N_K,q_K^f)$ on $E_A\cup E_B$.  Repeated application
constructs the root DtN map before any external boundary law is solved.

\section{Jacobi strong Petrov--Galerkin residual and face skeleton}

Let $K=F_K(\widehat K)$, $x=x_{K,0}+B_K\widehat x$, with $\widehat K$
the reference $D$-simplex.  Set
\begin{equation}\label{eq:full_trial_spaces}
 V_n(K)=\Pp_n^D(K),\qquad
 W_R(K)=\Pp_R^D(K),\qquad
 \Lambda_n(F)=\Pp_n^{D-1}(F),
\end{equation}
with $M=\binom{n+D}{D}$ volume modes, $m_R=\binom{R+D}{D}$
strong-residual modes, and $k_F=\binom{n+D-1}{D-1}$ modes per face.
The current full-degree elliptic implementation chooses $R=n$.
We represent $u_h\in V_n(K)$ by the orthonormal Jacobi-simplex basis
$\{\phi_j^{\kappa}\}$, and test the strong PDE residual against
$\{\psi_i^{\kappa_{\mathrm{res}}}\}$, with
\begin{equation}\label{eq:full_promoted}
 \kappa_{\mathrm{res}}=\kappa+\bm 2,
 \qquad w_\kappa(x)=
 \lambda_0^{\kappa_D-\frac12}
 \prod_{j=1}^{D}\lambda_j^{\kappa_{j-1}-\frac12},
 \qquad \kappa_j>-\tfrac12.
\end{equation}
The base parameter $\kappa_j=1/2$ gives the unweighted physical
polynomial family; general admissible $\kappa$ is allowed.
The canonical $(D-1)$-simplex face basis is fixed consistently on
all faces, including their orientations.

For $u_h=\sum_j a_{K,j}\phi_j^\kappa$ define the strong Petrov--Galerkin
moments
\begin{equation}\label{eq:full_moments}
 \R_{K,i}(u_h;f)=|\det B_K|\int_{\widehat K}
  \left[(\mathcal L_Ku_h-f)\circ F_K\right]
  \psi_i^{\kappa_{\mathrm{res}}}
  w_{\kappa_{\mathrm{res}}}\,\dd\widehat x.
\end{equation}
Write these moments as
\begin{equation}\label{eq:full_Lf}
 \bm r_K=L_Ka_K-\bm f_K,
\end{equation}
where the strong matrix $L_K$ and the promoted source moments
$\bm f_K$ use exactly the same residual family, geometry factors, and
quadrature convention.  In code paths that store the residual as
$L_Ka_K+f_{\mathrm{int},K}$, one has
$f_{\mathrm{int},K}=-\bm f_K$ with this continuum sign convention.

The divergence-form adapter is permitted to construct $L_K$ from the
expanded strong coefficients
\begin{equation}\label{eq:full_expanded}
 -A_{ij}\partial_{ij}u
 +\left(b_j-\sum_i\partial_iA_{ij}\right)\partial_ju+cu,
\end{equation}
using the existing Jacobi differentiation, promotion, and
multiplication machinery.  No weak-form stiffness matrix is substituted.

\vspace{0.5cm}

Let $T_K\in\mathbb R^{n_{\partial K}\times M}$ be the stacked physical
face trace map and $F_K\in\mathbb R^{n_{\partial K}\times M}$
the outward \emph{physical co-normal moment} map.  The common-skeleton
vector $\lambda_K\in\mathbb R^{n_{\partial K}}$ and
$T_Ka_K$ use the same canonical face coordinate representation.
The face space is orthonormal in the selected face weight (or must be
paired with an explicit face Riesz matrix); we keep these normalizations
visible when using physical unweighted boundary estimates.

\subsection{Weighted residuals can be converted to physical norms}
\label{sec:full_weighted_conversion}
The promoted family does not prevent us from estimating the physical
strong residual.  Two orthonormal Jacobi bases of the \emph{same}
polynomial space $\Pp_R^D$ are related by an invertible finite
connection/ladder matrix.  Let
\begin{equation}\label{eq:full_connection}
 C_{0\leftarrow\mathrm{res}}:
 [p]_{\kappa_{\mathrm{res}}}\longmapsto
 [p]_{\kappa_0},\qquad \kappa_0=(1/2,\ldots,1/2).
\end{equation}
JPolyD's parameter-connection machinery (including the arbitrary-valid-
parameter mode of the \texttt{KMat} builder) supplies such a map in
principle; it need not be sparse.  For a polynomial residual with
promoted \emph{coefficient} vector $\rho$, physical $L^2$ integration
has an explicitly computable Gram matrix
\begin{equation}\label{eq:full_residual_gram}
 \|p\|_{L^2(K)}^2
 =\rho^\top G_{0\leftarrow\mathrm{res},K}\rho,
 \quad G_{0\leftarrow\mathrm{res},K}
 =C_{0\leftarrow\mathrm{res}}^\top M_{0,K}
  C_{0\leftarrow\mathrm{res}}.
\end{equation}
Here $M_{0,K}$ includes the known reference normalization and physical
Jacobian.  If $\bm r_K$ is stored as \emph{moments} rather than coefficients,
one first applies the residual-family Riesz/Gram inverse (and the
known geometry scale) to obtain $\rho$.  Consequently, the physical
and promoted norms on $\Pp_R^D$ are equivalent, with constants given
by the extreme eigenvalues of this computable positive matrix.  This
is an algebraic route to explicit finite-$n$ bounds, not an assumed
asymptotic norm equivalence.

For nonpolynomial or insufficiently resolved variable coefficients,
$L_Ka_K-\bm f_K$ describes a \emph{projected} strong residual.  The
connection map converts that projected polynomial exactly; it does
not control the unrepresented residual tail or quadrature aliasing.
Such terms need separate consistency bounds in a full convergence
argument.

\section{Discrete leaf: strong residual and stabilized trace extension}

For fixed skeleton data $\lambda_K$ and promoted source moments
$\bm f_K$, define the present local stacked least-squares extension
by
\begin{equation}\label{eq:full_leaf_min}
 \boxed{\displaystyle
 a_K=\arg\min_a\mathcal J_K(a;\lambda_K,\bm f_K),\qquad
 \mathcal J_K=
 \tfrac12s_K^2\|L_Ka-\bm f_K\|_2^2
 +\tfrac12\|T_Ka-\lambda_K\|_{W_K}^2.}
\end{equation}
The interior row scale $s_K>0$ and positive definite face penalty
$W_K=W_K^\top\succ0$ are discretization parameters.  The active
implementation takes $W_K$ diagonal in canonical orthonormal face
coordinates and uses a dense QR factorization of the stacked matrix
\begin{equation}\label{eq:full_leaf_stack}
 \mathbb A_K=
 \begin{bmatrix}s_KL_K\\W_K^{1/2}T_K\end{bmatrix},
 \qquad
 \mathbb d_K=
 \begin{bmatrix}s_K\bm f_K\\W_K^{1/2}\lambda_K\end{bmatrix}.
\end{equation}
Define
\begin{equation}\label{eq:full_H}
 H_K=s_K^2L_K^\top L_K+T_K^\top W_KT_K.
\end{equation}
The elementary well-posedness criterion is exact.
\begin{proposition}[Unique local polynomial extension]
\label{prop:full_leaf_unique}
For $s_K>0$ and $W_K\succ0$, the minimizer in
\eqref{eq:full_leaf_min} is unique if and only if
$\ker L_K\cap\ker T_K=\{0\}$.
\end{proposition}
\begin{proof}
For any $a$, $a^\top H_Ka=s_K^2\|L_Ka\|_2^2+\|T_Ka\|_{W_K}^2$.
It vanishes precisely on the stated kernel intersection.  Therefore
$H_K\succ0$ exactly when the stacked system has full column rank.
\end{proof}
The normal equations yield an affine volume response
\begin{equation}\label{eq:full_Ulambda_cf}
 \boxed{\displaystyle
 a_K=U_{\lambda,K}\lambda_K+c_{f,K},\qquad
 U_{\lambda,K}=H_K^{-1}T_K^\top W_K,\qquad
 c_{f,K}=U_{f,K}\bm f_K,\quad
 U_{f,K}=H_K^{-1}s_K^2L_K^\top.}
\end{equation}
These inverses denote linear solves; the dense QR path computes the
multiple right-hand sides without explicitly forming $H_K^{-1}$.
In particular, $U_{\lambda,K}$ and $U_{f,K}$ can be cached and reused
for multiple source functions.

\subsection{Reduced least-squares energy and its boundary gradient}
\label{sec:full_reduced_energy}
Let $\mathcal J_{K,\mathrm{red}}(\lambda_K;\bm f_K)$ be
\eqref{eq:full_leaf_min} evaluated at its unique minimizer.
Define
\begin{equation}\label{eq:full_P}
 \boxed{\displaystyle
 P_K=W_K-W_KT_KU_{\lambda,K}
 =W_K-W_KT_KH_K^{-1}T_K^\top W_K.}
\end{equation}
\begin{proposition}[Positive reduced residual boundary operator]
\label{prop:full_P}
The matrix $P_K$ is symmetric positive semidefinite.  For zero source,
\begin{equation}\label{eq:full_P_energy}
 \lambda_K^\top P_K\lambda_K
 =\min_a\left(s_K^2\|L_Ka\|_2^2
 +\|T_Ka-\lambda_K\|_{W_K}^2\right)\ge0,
\end{equation}
and
\begin{equation}\label{eq:full_P_kernel}
 \ker P_K=T_K(\ker L_K).
\end{equation}
For arbitrary source, the exact reduced gradient is
\begin{equation}\label{eq:full_P_gradient}
 \boxed{\displaystyle
 \nabla_{\lambda_K}\mathcal J_{K,\mathrm{red}}
 =-W_K(T_Ka_K-\lambda_K)
 =P_K\lambda_K-W_KT_Kc_{f,K}.}
\end{equation}
\end{proposition}
\begin{proof}
The normal equations give
$H_Ka_K=T_K^\top W_K\lambda_K+s_K^2L_K^\top\bm f_K$.
Inserting $a_K=U_{\lambda,K}\lambda_K$ into the homogeneous
quadratic objective gives \eqref{eq:full_P_energy}.  The explicit
matrix form is symmetric since $H_K$ and $W_K$ are symmetric.
Positivity follows from the minimum of two squares.  Equality in the
homogeneous minimum is equivalent to an $a$ with $L_Ka=0$ and
$T_Ka=\lambda_K$, which establishes the kernel identity.  Finally,
differentiating the objective with respect to fixed $\lambda_K$ at
its stationary minimizer eliminates the $\partial a_K/\partial\lambda_K$
term and gives \eqref{eq:full_P_gradient}.\end{proof}

\subsection{Corrected augmented outward co-normal map}
\label{sec:full_augmented}
Write the mismatch as $\delta_K=T_Ka_K-\lambda_K$.
The numerical outward co-normal \emph{moment} vector is defined by
\begin{equation}\label{eq:full_mu}
 \boxed{\displaystyle
 \mu_K=F_Ka_K-W_K(T_Ka_K-\lambda_K)
 =F_Ka_K+\nabla_{\lambda_K}\mathcal J_{K,\mathrm{red}}.}
\end{equation}
The physical co-normal map $F_K$ uses the original divergence tensor
$A_K$ and the outward normal, not the expanded strong operator's
first-order coefficient.  The augmentation is thus exactly the
skeleton derivative of the minimized residual energy.  This fact
justifies its sign but does \emph{not} assert that the entire flux
$F_Ka_K$ is itself a gradient of a scalar functional.

Inserting \eqref{eq:full_Ulambda_cf} yields the complete affine leaf map
\begin{equation}\label{eq:full_leaf_Sb}
 \boxed{\begin{aligned}
 \mu_K&=S_K\lambda_K+b_K,\\[-1mm]
 S_K&=(F_K-W_KT_K)U_{\lambda,K}+W_K
      =F_KU_{\lambda,K}+P_K,\\
 b_K&=(F_K-W_KT_K)c_{f,K}
      =(F_K-W_KT_K)U_{f,K}\bm f_K.
 \end{aligned}}
\end{equation}
Here $S_K$ is the moment-space \emph{discrete Steklov/DtN
approximation} corresponding to the continuous $N_K$, and $b_K$ is
the discrete source-induced co-normal response corresponding to
$q_K^f$.  They are not identically equal to their continuum analogues
at finite polynomial order.  The discrete map acts on the complete
per-element polynomial skeleton, including skeleton traces that do not
extend to exact homogeneous polynomial solutions.  $P_K$ controls
precisely that incompatibility.

For zero source, $S_K=F_KU_{\lambda,K}+P_K$.  In particular,
with $\operatorname{sym}(\cdot)$ the symmetric part,
\begin{equation}\label{eq:full_Ssym}
 \operatorname{sym}(S_K)
 =\operatorname{sym}(F_KU_{\lambda,K})+P_K.
\end{equation}
The PSD conclusion applies to $P_K$, not automatically to $S_K$.
For nonsymmetric advection operators, $F_KU_{\lambda,K}$ can be
nonsymmetric and its symmetric part need not be nonnegative.  Global
stability and invertibility of merge interface matrices require
additional arguments or hypotheses.

\subsection{A conditional PDE energy route, including reaction and advection}
The continuous Green identity for a local polynomial $u_h$ is
\begin{equation}\label{eq:full_green_energy}
 \dual{q_K(u_h)}{u_h}_{\partial K}
 =\|A_K^{1/2}\nabla u_h\|_{L^2(K)}^2
 +(c_{\mathrm{eff},K}u_h,u_h)_K
 +\frac12\dual{\bm b_K\cdot\bm n_K}{u_h^2}_{\partial K}
 -(\mathcal L_Ku_h,u_h)_K,
\end{equation}
where $c_{\mathrm{eff},K}=c_K-\frac12\nabla\cdot\bm b_K$.
For $\bm f_K=0$, the exact least-squares normal equations give
\begin{equation}\label{eq:full_normal_energy}
 -\lambda_K^\top W_K\delta_K
 =s_K^2\|L_Ka_K\|_2^2+\|\delta_K\|_{W_K}^2.
\end{equation}
Combining this with $\lambda_K=T_Ka_K-\delta_K$ and the physical
Green identity gives, subject to proper face Riesz conversion,
\begin{align}\label{eq:full_energy_route}
 \dual{\mu_K}{\lambda_K}_{\partial K}
 ={}&\|A^{1/2}\nabla u_h\|_{L^2(K)}^2
 +(c_{\mathrm{eff}}u_h,u_h)_K
 +\tfrac12\dual{\bm b\cdot\bm n}{u_h^2}_{\partial K}
 \nonumber\\
 &+s_K^2\|L_Ka_K\|_2^2+\|\delta_K\|_{W_K}^2
 -\dual{q_K(u_h)}{\delta_K}_{\partial K}
 -(\mathcal Lu_h,u_h)_K.
\end{align}
The last two terms are not sign-definite; the first is estimated using
face trace/inverse inequalities and a diffusion-scaled $W_K$, and the
second using the physical-$L^2$ conversion in
Section~\ref{sec:full_weighted_conversion}, plus estimates for any
unresolved residual tail.  For sufficiently positive effective
reaction the bulk term is helpful; no additional \emph{physical}
reaction flux occurs on faces.  Advection enters the boundary energy
and may motivate a directional face weighting if the baseline penalty
is insufficient, but is not postulated as mandatory here.
Equation \eqref{eq:full_energy_route} is a \emph{proof strategy}, not a
claimed unconditional coercivity theorem.  It also makes no hidden
assumption that the entire HPS flux comes from a global least-squares
Euler--Lagrange principle.

\section{Exact algebraic HPS merge of two discrete subdomains}
\label{sec:full_merge}

Consider two already-built (possibly composite) child HPS nodes
$A$ and $B$, with boundary data and augmented outward co-normal maps
\begin{equation}\label{eq:full_two_children}
 \mu_A=S_A\lambda_A+b_A,\qquad
 \mu_B=S_B\lambda_B+b_B.
\end{equation}
The skeleton is single-valued across the shared interface.  Denote
by $E_A$ and $E_B$ the child faces exterior to their parent, and by
$I$ their common set of interface faces.  In the canonical orientation
and ordering, write
\begin{equation}\label{eq:full_child_traces}
 \lambda_A=\begin{bmatrix}\lambda_{A,E}\\\gamma\end{bmatrix},
 \qquad
 \lambda_B=\begin{bmatrix}\lambda_{B,E}\\\gamma\end{bmatrix}.
\end{equation}
Physical normals on the interface are opposite, but the interface
basis coefficients $\gamma$ have been mapped to the \emph{same}
canonical face basis before merging.  This is precisely what allows
fluxes to add with their outward signs.

Partition the child maps into exterior and interface blocks:
\begin{align}\label{eq:full_child_blockA}
 \begin{bmatrix}\mu_{A,E}\\\mu_{A,I}\end{bmatrix}
 &=\begin{bmatrix}S_{A,EE}&S_{A,EI}\\S_{A,IE}&S_{A,II}\end{bmatrix}
 \begin{bmatrix}\lambda_{A,E}\\\gamma\end{bmatrix}
 +\begin{bmatrix}b_{A,E}\\b_{A,I}\end{bmatrix},\\
 \begin{bmatrix}\mu_{B,E}\\\mu_{B,I}\end{bmatrix}
 &=\begin{bmatrix}S_{B,EE}&S_{B,EI}\\S_{B,IE}&S_{B,II}\end{bmatrix}
 \begin{bmatrix}\lambda_{B,E}\\\gamma\end{bmatrix}
 +\begin{bmatrix}b_{B,E}\\b_{B,I}\end{bmatrix}.
 \label{eq:full_child_blockB}
\end{align}
Only the interface flux moments are eliminated.  The parent exterior
trace and outward co-normal numerical flux are
\begin{equation}\label{eq:full_parent_trace}
 \lambda_P=\begin{bmatrix}\lambda_{A,E}\\\lambda_{B,E}\end{bmatrix},
 \qquad
 \mu_P=\begin{bmatrix}\mu_{A,E}\\\mu_{B,E}\end{bmatrix}.
\end{equation}

\subsection{Interface continuity and the reconstruction map}
The HPS interface balance is
\begin{equation}\label{eq:full_flux_balance}
 \mu_{A,I}+\mu_{B,I}=0.
\end{equation}
Inserting the interface rows of \eqref{eq:full_child_blockA}--
\eqref{eq:full_child_blockB} gives
\begin{equation}\label{eq:full_interface_H}
 \underbrace{(S_{A,II}+S_{B,II})}_{H_I}\gamma
 =-S_{A,IE}\lambda_{A,E}-S_{B,IE}\lambda_{B,E}
  -b_{A,I}-b_{B,I}.
\end{equation}
Assume $H_I$ is invertible.  Define
\begin{equation}\label{eq:full_RAr}
 \boxed{\begin{aligned}
 R_A&=-H_I^{-1}S_{A,IE},\\
 R_B&=-H_I^{-1}S_{B,IE},\\
 r&=-H_I^{-1}(b_{A,I}+b_{B,I}).
 \end{aligned}}
\end{equation}
Then the downward reconstruction equation for the interior interface
skeleton is
\begin{equation}\label{eq:full_gamma}
 \boxed{\gamma=R_A\lambda_{A,E}+R_B\lambda_{B,E}+r.}
\end{equation}
This is an algebraic Schur elimination and does not require the
operator to be self-adjoint.  For advective elliptic problems the
matrix $H_I$ may be nonsymmetric; invertibility and conditioning
are properties to check, not consequences of $P_K\succeq0$ alone.

\subsection{Parent homogeneous and source-induced maps}
Substitute \eqref{eq:full_gamma} into the \emph{external} rows of
\eqref{eq:full_child_blockA}--\eqref{eq:full_child_blockB}:
\begin{equation}\label{eq:full_parent_flux_compact}
 \begin{bmatrix}\mu_{A,E}\\\mu_{B,E}\end{bmatrix}
 =\begin{bmatrix}S_{A,EE}&0\\0&S_{B,EE}\end{bmatrix}
 \begin{bmatrix}\lambda_{A,E}\\\lambda_{B,E}\end{bmatrix}
 +\begin{bmatrix}S_{A,EI}\\S_{B,EI}\end{bmatrix}
 (R_A\lambda_{A,E}+R_B\lambda_{B,E}+r)
 +\begin{bmatrix}b_{A,E}\\b_{B,E}\end{bmatrix}.
\end{equation}
Therefore the parent again has exactly the same affine Steklov form:
\begin{equation}\label{eq:full_parent_Sb}
 \boxed{\mu_P=S_P\lambda_P+b_P,}
\end{equation}
where
\begin{equation}\label{eq:full_parent_S_block}
 \boxed{\displaystyle
 S_P=\begin{bmatrix}
 S_{A,EE}+S_{A,EI}R_A&S_{A,EI}R_B\\
 S_{B,EI}R_A&S_{B,EE}+S_{B,EI}R_B
 \end{bmatrix},}
\end{equation}
and
\begin{equation}\label{eq:full_parent_b_block}
 \boxed{\displaystyle
 b_P=\begin{bmatrix}
 b_{A,E}+S_{A,EI}r\\
 b_{B,E}+S_{B,EI}r
 \end{bmatrix}.}
\end{equation}
This is the full HPS merge, including the affine source contribution.
The parent has the same data type as either child: a face-only
operator $S_P$, a source-dependent shift $b_P$, and the data needed
to reconstruct $\gamma$.  Recursively, two triangles, tetrahedra, or
arbitrary simplex subdomains build the root map using precisely the
same formula.

\subsection{Reusable source-only upward pass}
The expensive matrices $S_A,S_B,H_I,R_A,R_B$ depend on the
operator and geometry, not on a particular source vector.  Define
\begin{equation}\label{eq:full_source_inverse}
 R_f:=-H_I^{-1},\qquad
 P_A:=S_{A,EI}R_f,\qquad P_B:=S_{B,EI}R_f.
\end{equation}
For a new source, once child shifts are known, update
\begin{align}\label{eq:full_source_pass}
 r&=R_f(b_{A,I}+b_{B,I}),\\
 b_P&=\begin{bmatrix}
 b_{A,E}+P_A(b_{A,I}+b_{B,I})\\
 b_{B,E}+P_B(b_{A,I}+b_{B,I})
 \end{bmatrix}.
\end{align}
This upward pass reuses the stored operators without repeating the
matrix factorization or rebuilding the homogeneous DtN maps.  The
implementation stores $R_f,P_A,P_B$ for this reason.  At each leaf,
a new volume source only updates
$c_{f,K}=U_{f,K}\bm f_K$ and
$b_K=(F_K-W_KT_K)c_{f,K}$.

\section{Root Robin solve and downward recovery}
\label{sec:full_root}
At the root, the hierarchy has produced
\begin{equation}\label{eq:full_root_map}
 \mu_{\mathrm{root}}
 =S_{\mathrm{root}}\lambda_{\mathrm{root}}
 +b_{\mathrm{root}}.
\end{equation}
Consider a continuous boundary law
\begin{equation}\label{eq:full_root_Robin}
 \alpha(x)u+\beta(x)q(u)=g(x)
 \qquad\text{on }\partial\Omega.
\end{equation}
In the canonical root face moment representation, let
$M_\alpha,M_\beta$ denote the appropriate face multiplication and
projection operators.  Imposing the Robin law on the numerical
co-normal flux gives
\begin{equation}\label{eq:full_root_Robin_system}
 \boxed{\displaystyle
 (M_\alpha+M_\beta S_{\mathrm{root}})
 \lambda_{\mathrm{root}}
 =\bm g-M_\beta b_{\mathrm{root}}.}
\end{equation}
For constant facewise coefficients these are scalar blocks.  For
variable coefficients, consistent integration of $\beta q$ may require
a higher-degree co-normal product correction when multiplication and
face projection do not commute (see
\texttt{high\_order\_conormal\_robin\_hps.tex} for dense materialization
of the multiplication operator and high order correction details).
The abstract root formula
\eqref{eq:full_root_Robin_system} is understood to include that
consistent moment operation.  Because $\mu$ is augmented, a numerical
Robin law generally differs from pointwise enforcement of the raw
physical co-normal flux at finite $n$; the discrepancy is a validation
diagnostic, not an algebraic contradiction.

Dirichlet conditions correspond to $\beta=0$ and determine the trace;
mixed/Robin conditions use the corresponding rows.  In the pure
Neumann Laplace limit, constants lie in the kernel; compatibility and
a gauge constraint are necessary.  For general advection/reaction
operators, compatibility must be derived from that operator and must
not be replaced mechanically by the Laplace formula (see
\texttt{pure\_neumann\_jpolyd\_hps.tex} for details).

\paragraph{Downward skeleton reconstruction.}
After solving the root boundary system, traverse the merge tree from
root to leaves.  For every parent split, partition its recovered
exterior trace as $(\lambda_{A,E},\lambda_{B,E})$, evaluate
\begin{equation}\label{eq:full_downward_gamma}
 \gamma=R_A\lambda_{A,E}+R_B\lambda_{B,E}+r,
\end{equation}
and assemble each child's complete trace
$\lambda_A=(\lambda_{A,E},\gamma)$,
$\lambda_B=(\lambda_{B,E},\gamma)$.
Continue recursively until all leaf skeleton traces are available.

\paragraph{Downward volume reconstruction.}
At each leaf, recover the Jacobi modal coefficients from the cached
affine response:
\begin{equation}\label{eq:full_reconstruction}
 \boxed{a_K=U_{\lambda,K}\lambda_K+c_{f,K}.}
\end{equation}
The physical solution is then evaluated using the library's Jacobi
basis evaluators, and physical gradients with their affine Jacobian
maps.  If desired, $\mu_K=S_K\lambda_K+b_K$ and the separately
reconstructed raw co-normal moments $F_Ka_K$ can be compared to
monitor the trace-mismatch flux correction.

\paragraph{Complete computational ordering.}
For a fixed operator: (i) construct leaf $U_{\lambda},U_f,S$ and
source-response maps; (ii) merge upward, storing interface inverses
and reconstruction maps; (iii) for each new source, compute leaf
$c_f,b$ and propagate $b$ and $r$ upward; (iv) solve root Robin data;
(v) reconstruct all internal traces downward; and (vi) reconstruct
volume modal coefficients.  The outward augmented co-normal penalty
appears only in the leaf maps; all later HPS levels manipulate their
affine face operators.

\section{Algebraic verification and limits of the stability statement}
The bundled \texttt{verify\_pk.py} creates a full-rank illustrative
leaf (or accepts exported $L,T,W,s$ in an NPZ file), constructs
\begin{equation}\label{eq:full_verify_def}
 H=s^2L^\top L+T^\top WT,\quad
 U_\lambda=H^{-1}T^\top W,\quad
 P=W-WTU_\lambda,
\end{equation}
and checks: (a) $P=P^\top$ to roundoff; (b) the eigenvalues of
$\operatorname{sym}(P)$ are nonnegative up to roundoff;
(c) $\lambda^\top P\lambda$ equals the minimized homogeneous
residual energy; and (d) the source-dependent gradient identity
\eqref{eq:full_P_gradient}.  These are exact algebraic identities
under the rank assumption and do not depend on spatial dimension,
Jacobi weights, or PDE coefficients.  The companion script
\texttt{verify\_merge.py} independently checks the two-child
interface balance and reconstructed parent affine map using
synthetic nonsymmetric child blocks, including nonzero source shifts.

For the \emph{complete} discrete Steklov map, positivity of $P$ does
not by itself prove coercivity of $S=FU_\lambda+P$, uniqueness of
all merge interface solves, preservation of a maximum principle, or
uniform high-P\'eclet convergence.  A full theorem additionally
requires estimates for the physical co-normal response and its
trace mismatch, the conversion of the promoted projected residual
to a physical norm (Section~\ref{sec:full_weighted_conversion}),
and any unrepresented coefficient/quadrature residual.  Positive
$c-\frac12\nabla\cdot\bm b$ contributes helpful bulk control;
reaction is not an independent physical face flux.  An inflow-weighted
penalty may be explored later if the baseline method fails outside
the tested parameter range.  None of these open analytical questions
requires replacing the strong-form Jacobi operator or the HPS merge
algebra.

\subsection*{Block sparsity assumptions on Jacobi simplex operators}
	
	Let
	\[
	\Pi_n^D = \bigoplus_{j=0}^n \mathcal{H}_j^D
	\]
	denote the decomposition of total-degree polynomials into homogeneous subspaces
	\(\mathcal{H}_j^D\) of degree exactly \(j\).
	We fix, for each \(j\), a Jacobi simplex basis
	\(\{\phi^{(\kappa)}_{\alpha} : |\alpha|=j\}\) of \(\mathcal{H}_j^D\),
	ordered by a graded rule.
	We assume the existence of a canonical bijection between multi-indices
	\(\alpha\) with \(|\alpha|=j\) and local indices in \(\mathcal{H}_j^D\).
	
	All operators below are represented in this basis.
	
	\paragraph{(i) Degree-lowering operators (DMat).}
	
	Let
	\[
	\mathcal{D} : \Pi_n^D \to \Pi_{n-1}^D
	\]
	be a linear operator such that
	\[
	\mathcal{D}\,\mathcal{H}_j^D \subset \mathcal{H}_{j-1}^D,
	\qquad j \ge 1 .
	\]
	
	Then the matrix of \(\mathcal{D}\) has block structure
	\[
	\mathcal{D}
	=
	\begin{pmatrix}
		0      & 0      & 0      & \cdots & 0 \\
		D_1    & 0      & 0      & \cdots & 0 \\
		0      & D_2    & 0      & \cdots & 0 \\
		\vdots & \ddots & \ddots & \ddots & \vdots \\
		0      & \cdots & 0      & D_n    & 0
	\end{pmatrix},
	\]
	where \(D_j : \mathcal{H}_j^D \to \mathcal{H}_{j-1}^D\).
	
	\emph{Sparsity assumption.}
	There exists a finite set
	\[
	\Delta_{\mathcal{D}} \subset \mathbb{Z}^D,
	\qquad \sum_{d=1}^D \Delta_d = -1,
	\]
	independent of \(j\), such that
	\[
	\mathcal{D}\phi^{(\kappa)}_{\alpha}
	\in
	\operatorname{span}
	\bigl\{
	\phi^{(\kappa')}_{\alpha+\Delta} : \Delta \in \Delta_{\mathcal{D}}
	\bigr\}.
	\]
	
\paragraph{(ii) Degree-filtering operators (KMat).}

Let
\[
\mathcal{K} : \Pi_n^D \to \Pi_n^D
\]
be a linear operator arising from a \emph{natural promotion} of Jacobi parameters,
\[
\kappa_{\mathrm{tgt}} = \kappa_{\mathrm{src}} + e_r,
\qquad r \in \{0,\dots,D\},
\]
corresponding to multiplication of the orthogonality weight by a single barycentric factor.
Although $\mathcal{K}$ preserves the total polynomial degree globally, it is \emph{not}
degree-preserving at the level of homogeneous subspaces.

More precisely, $\mathcal{K}$ satisfies the block filtering property
\[
\mathcal{K}\,\mathcal{H}_j^D
\subset
\mathcal{H}_j^D \;\oplus\; \mathcal{H}_{j-1}^D,
\qquad j \ge 0,
\]
with the convention $\mathcal{H}_{-1}^D = \{0\}$.
Equivalently, the matrix of $\mathcal{K}$ in the graded basis
\[
\Pi_n^D = \bigoplus_{j=0}^n \mathcal{H}_j^D
\]
has a two-band block structure:
\[
\mathcal{K}
=
\begin{pmatrix}
	A_0 & B_0 & 0   & 0   & \cdots \\
	0   & A_1 & B_1 & 0   & \cdots \\
	0   & 0   & A_2 & B_2 & \cdots \\
	\vdots & & \ddots & \ddots & \ddots
\end{pmatrix},
\]
where
\[
A_j : \mathcal{H}_j^D \to \mathcal{H}_j^D,
\qquad
B_j : \mathcal{H}_{j+1}^D \to \mathcal{H}_j^D .
\]
That is, writing $u$ as a graded decomposition $$u = \sum_{j=0}^n u_j,\quad u_j \in \mathcal{H}_j^{\kappa_{\text{src}}},$$ we define the operator $$\mathcal{K} : \Pi_n^{\kappa_{\text{src}}} \to \Pi_n^{\kappa_{\text{tgt}}},$$ so that $$\mathcal{K}u = \sum_{i=0}^n (\mathcal{K} u)_i,\quad (\mathcal{K}u)_i \in \mathcal{H}_i^{\text{tgt}}.$$ So, target block $i$ recieves contributions from source blocks $i$ and $i+1$. Now, if $u$ only has one homogeneous component, $u\equiv u_j \in \mathcal{H}_j^{\text{src}}$, then the only terms remaining in the sum over $i$ from the formula
\begin{align*}
(\mathcal{K} u)_i = A_iu_i + B_iu_{i+1}
\end{align*}
are $i=j$ via $A_j u_j$ and $i=j-1$ via $B_{j-1}u_j$. Then, we see that $$\mathcal{K} u_j = (\mathcal{K}u)_j + (\mathcal{K}u)_{j-1} \Rightarrow \mathcal{K} \mathcal{H}_j \subset \mathcal{H}_j \oplus \mathcal{H}_{j-1}.$$

\medskip
\emph{Sparsity assumption.}
There exist finite sets of multi-index offsets
\[
\Delta^{(0)}_{\mathcal{K}},\;
\Delta^{(-1)}_{\mathcal{K}}
\subset \mathbb{Z}^D,
\]
independent of $j$, such that for every basis function
$\phi^{(\kappa_{\mathrm{src}})}_{\alpha}$ with $|\alpha| = j$,
\[
\mathcal{K}\phi^{(\kappa_{\mathrm{src}})}_{\alpha}
\in
\operatorname{span}
\Bigl(
\{\phi^{(\kappa_{\mathrm{tgt}})}_{\alpha+\Delta} : \Delta \in \Delta^{(0)}_{\mathcal{K}}\}
\;\cup\;
\{\phi^{(\kappa_{\mathrm{tgt}})}_{\alpha+\Delta} : \Delta \in \Delta^{(-1)}_{\mathcal{K}},\ |\alpha+\Delta|=j-1\}
\Bigr).
\]
The sets $\Delta^{(0)}_{\mathcal{K}}$ and $\Delta^{(-1)}_{\mathcal{K}}$ encode,
respectively, the intra-degree couplings ($A_j$ blocks) and the one-degree-down couplings
($B_j$ blocks).

\medskip
\emph{Stencil discovery.}
To determine these coupling patterns, we assemble a dense reference operator
$\mathcal{K}^{(n_{\mathrm{test}})}$ at a small test degree $n_{\mathrm{test}}$
and restrict it to the representative row block
\[
\mathcal{K}^{(n_{\mathrm{test}})}_{\mathrm{rep}}
=
\mathcal{K}^{(n_{\mathrm{test}})}
\big|_{\mathcal{H}^D_{j_{\mathrm{rep}}}
	\times
	\left(\mathcal{H}^D_{j_{\mathrm{rep}}} \oplus \mathcal{H}^D_{j_{\mathrm{rep}}+1}\right)},
\qquad j_{\mathrm{rep}} = n_{\mathrm{test}} - 1 .
\]
For each nonzero entry in this restricted block, we record the multi-index difference
\[
\Delta = \alpha_{\mathrm{row}} - \alpha_{\mathrm{col}},
\]
partitioning the resulting offsets according to whether they correspond to
$\mathcal{H}_j \leftarrow \mathcal{H}_j$ or
$\mathcal{H}_j \leftarrow \mathcal{H}_{j+1}$ couplings.
Once these offset sets stabilize as $n_{\mathrm{test}}$ increases, they are reused
to assemble $\mathcal{K}$ at arbitrary polynomial degree $n$ without further dense computation.

	
	\paragraph{(iii) Three-term recurrence operators (JMat).}
	
	Let
	\[
	\mathcal{J} : \Pi_n^D \to \Pi_{n+1}^D
	\]
	be a linear operator satisfying
	\[
	\mathcal{J}\,\mathcal{H}_j^D
	\subset
	\mathcal{H}_{j-1}^D \oplus \mathcal{H}_j^D \oplus \mathcal{H}_{j+1}^D .
	\]
	
	Then the matrix of \(\mathcal{J}\) has block tridiagonal structure:
	\[
	\mathcal{J}
	=
	\begin{pmatrix}
		B_0 & C_0 & 0   & \cdots & 0 \\
		A_1 & B_1 & C_1 & \ddots & \vdots \\
		0   & A_2 & B_2 & \ddots & 0 \\
		\vdots & \ddots & \ddots & \ddots & C_{n-1} \\
		0 & \cdots & 0 & A_n & B_n
	\end{pmatrix},
	\]
	with
	\[
	A_j : \mathcal{H}_j^D \to \mathcal{H}_{j-1}^D,\quad
	B_j : \mathcal{H}_j^D \to \mathcal{H}_j^D,\quad
	C_j : \mathcal{H}_j^D \to \mathcal{H}_{j+1}^D .
	\]
	
	\emph{Sparsity assumption.}
	There exists a finite set
	\[
	\Delta_{\mathcal{J}} \subset \mathbb{Z}^D,
	\qquad \sum_{d=1}^D \Delta_d \in \{-1,0,+1\},
	\]
	independent of \(j\), such that
	\[
	\mathcal{J}\phi^{(\kappa)}_{\alpha}
	\in
	\operatorname{span}
	\bigl\{
	\phi^{(\kappa')}_{\alpha+\Delta} : \Delta \in \Delta_{\mathcal{J}}
	\bigr\}.
	\]
	\subsection*{Delta-based representation of operator couplings}
	
	Let $\mathcal{O}$ be a linear operator acting on the graded polynomial space
	\[
	\Pi_n^D = \bigoplus_{j=0}^n \mathcal{H}_j^D,
	\]
	where $\mathcal{H}_j^D$ denotes the subspace of homogeneous polynomials of total
	degree $j$ in $D$ variables.
	Fix a Jacobi simplex basis
	$\{\phi^{(\kappa)}_{\alpha} : |\alpha| = j\}$ for each $\mathcal{H}_j^D$,
	indexed by multi-indices $\alpha \in \mathbb{N}^D$ with $|\alpha|=j$.
	
	\paragraph{Coupling types.}
	We assume that the action of $\mathcal{O}$ on a basis function
	$\phi^{(\kappa)}_{\alpha}$ can be written in the form
	\[
	\mathcal{O}\phi^{(\kappa)}_{\alpha}
	=
	\sum_{\Delta \in \Delta_{\mathcal{O}}}
	c_{\Delta}(\alpha)\,
	\phi^{(\kappa')}_{\alpha+\Delta},
	\]
	where:
	\begin{itemize}
		\item $\Delta_{\mathcal{O}} \subset \mathbb{Z}^D$ is a finite set of
		\emph{multi-index shifts} (``deltas''),
		\item the coefficients $c_{\Delta}(\alpha)$ may depend on $\alpha$ and the
		Jacobi parameters, but
		\item the set $\Delta_{\mathcal{O}}$ itself is independent of the polynomial
		degree $j$.
	\end{itemize}
	
	Each $\Delta \in \Delta_{\mathcal{O}}$ represents a distinct \emph{coupling type}
	between a source basis function indexed by $\alpha$ and a target basis function
	indexed by $\alpha+\Delta$.
	
	\paragraph{Admissibility.}
	For a fixed source multi-index $\alpha$, a shift $\Delta$ is said to be
	\emph{admissible} if the target multi-index
	\[
	\beta = \alpha + \Delta
	\]
	satisfies:
	\begin{enumerate}
		\item componentwise nonnegativity, $\beta_i \ge 0$ for all $i$, and
		\item the appropriate degree constraint
		\[
		|\beta| = |\alpha| + s,
		\]
		where $s$ is the degree shift associated with $\mathcal{O}$.
	\end{enumerate}
	Only admissible shifts contribute to the action of the operator at a given
	$\alpha$.
	
	\paragraph{Stability of the delta set.}
	For small degrees $j$, some coupling types may be absent due to boundary effects
	in the index set $\{ \alpha : |\alpha| = j \}$.
	However, there exists a degree $j_\ast$ such that for all $j \ge j_\ast$ the set
	of observed admissible shifts coincides with a fixed finite set
	$\Delta_{\mathcal{O}}$.
	We refer to this property as \emph{stability} of the delta set.
	
	\paragraph{Assembly at arbitrary degree.}
	Once a stable delta set $\Delta_{\mathcal{O}}$ has been identified, the matrix
	representation of $\mathcal{O}$ on $\Pi_n^D$ for arbitrary $n$ may be constructed
	as follows:
	\begin{enumerate}
		\item loop over all basis functions (columns) indexed by $\alpha$,
		\item for each $\alpha$, loop over all $\Delta \in \Delta_{\mathcal{O}}$,
		\item form the candidate target index $\beta = \alpha + \Delta$,
		\item if $\beta$ is admissible, insert the corresponding matrix entry
		coupling $\phi^{(\kappa)}_{\alpha}$ to $\phi^{(\kappa')}_{\beta}$.
	\end{enumerate}
	The mapping between multi-indices and linear matrix indices is provided by the
	fixed ordering of the Jacobi basis within each homogeneous subspace and depends
	on the total degree, but the set of admissible shifts $\Delta_{\mathcal{O}}$
	is independent of $n$.
	
	\paragraph{Interpretation.}
	In this formulation, the delta set $\Delta_{\mathcal{O}}$ encodes all possible
	local coupling types induced by the operator in multi-index space.
	The degree dependence of the matrix sparsity pattern arises solely from the
	admissibility conditions and the enumeration of basis functions, not from changes in the underlying operator structure. The apparent spreading of nonzeros with increasing degree is a consequence of the growing number of basis functions between two fixed multi-index shifts in the graded ordering; the underlying multi-index coupling encoded by each delta remains unchanged.

\subsection*{Data-driven discovery of the finite delta sets $\Delta_\ast$}

For each operator family $\mathcal{O}$ (e.g.\ $\mathcal{D}$, $\mathcal{K}$, $\mathcal{J}$)
we assume the existence of a finite multi-index shift set
\[
\Delta_{\mathcal{O}} \subset \mathbb{Z}^D
\]
such that, for every homogeneous degree $j$ and every basis index $\alpha$ with $|\alpha|=j$,
the action of $\mathcal{O}$ on $\phi^{(\kappa)}_\alpha$ is supported only on shifts
$\alpha+\Delta$ with $\Delta \in \Delta_{\mathcal{O}}$ (cf.\ the sparsity assumptions above).
We describe a practical strategy to discover $\Delta_{\mathcal{O}}$.

\paragraph{Step 0: choose a representative homogeneous block.}
Fix a test degree $n_{\mathrm{test}}$ and select a representative block degree
\[
j_{\mathrm{rep}} = j_{\mathrm{rep}}(n_{\mathrm{test}})
\]
for extracting the stencil. In our implementation we take
\[
j_{\mathrm{rep}} = n_{\mathrm{test}}-1,
\]
i.e. the second highest-degree homogeneous block in $\Pi^D_{n_{\mathrm{test}}}$,
which is the largest block and empirically exhibits the full interior coupling.

\paragraph{Step 1: assemble a pruned dense test operator.}
Assemble a dense matrix representation
\[
O^{(n_{\mathrm{test}})} \in \mathbb{R}^{M\times M},
\qquad M=\dim \Pi_{n_{\mathrm{test}}}^D,
\]
for the operator $\mathcal{O}$ (with the desired parameters $\kappa$, shifts, and axis),
and apply numerical pruning to remove entries below a relative tolerance
(e.g.\ row-relative threshold proportional to machine epsilon). This produces
a matrix whose remaining nonzeros reflect the algebraic sparsity pattern.

\paragraph{Step 2: extract the set of observed deltas from the representative block.}
Restrict $O^{(n_{\mathrm{test}})}$ to the representative homogeneous block
\[
O^{(n_{\mathrm{test}})}_{\mathrm{rep}}
=
O^{(n_{\mathrm{test}})}\big|_{\mathcal{H}^D_{j_{\mathrm{rep}}}\times \mathcal{H}^D_{j_{\mathrm{rep}}+s}},
\]
where the degree shift $s$ is determined by the operator:
\[
s=
\begin{cases}
	-1, & \mathcal{O}=\mathcal{D} \\
	\in\{-1,0\},  & \mathcal{O}=\mathcal{K} \\
	\in\{-1,0,+1\}, & \mathcal{O}=\mathcal{J}\ \text{(handled per block band)}
\end{cases}
\]
and where $\mathcal{H}^D_{j_{\mathrm{rep}}+s}$ denotes the appropriate source block.
For each nonzero entry
\[
\bigl(O^{(n_{\mathrm{test}})}_{\mathrm{rep}}\bigr)_{i,j} \neq 0,
\]
let $\alpha_i$ and $\alpha_j$ be the (unique) multi-indices associated with the
row and column basis functions in the chosen ordering (existence assumed).
Define the observed delta
\[
\Delta = \alpha_i - \alpha_j \in \mathbb{Z}^D.
\]
Collect all such deltas and retain the unique set:
\[
\Delta^{(n_{\mathrm{test}})}_{\mathcal{O}}
=
\Bigl\{
\alpha_i - \alpha_j \,:\,
\bigl(O^{(n_{\mathrm{test}})}_{\mathrm{rep}}\bigr)_{i,j} \neq 0
\Bigr\}.
\]

\paragraph{Step 3: stabilization over increasing $n_{\mathrm{test}}$.}
Starting from a minimal degree $n_{\min}$ (e.g.\ $n_{\min}=5$), repeat Steps 1--2
for $n_{\mathrm{test}} = n_{\min}, n_{\min}+1, \dots, n_{\max}$, producing a sequence
of candidate delta sets $\Delta^{(n_{\mathrm{test}})}_{\mathcal{O}}$.
We declare the stencil to have stabilized when
\[
\Delta^{(n_{\mathrm{test}})}_{\mathcal{O}}
=
\Delta^{(n_{\mathrm{test}}-1)}_{\mathcal{O}}
\]
(as sets, e.g.\ after sorting a canonical encoding), and set
\[
\Delta_{\mathcal{O}} := \Delta^{(n_{\mathrm{test}})}_{\mathcal{O}}.
\]

\paragraph{Usage at arbitrary degree.}
Once $\Delta_{\mathcal{O}}$ is known, the operator at any target degree $n$
can be assembled by looping over source basis functions (multi-indices $\alpha$)
and applying the admissible shifts:
\[
\alpha \mapsto \alpha+\Delta, \qquad \Delta \in \Delta_{\mathcal{O}},
\]
discarding invalid targets (e.g.\ negative components or incorrect total degree),
and placing the corresponding nonzero entry in the sparse matrix representation.
The conversion between multi-indices and local indices within $\mathcal{H}_j^D$
is performed by fixed, degree-consistent maps induced by the chosen ordering.

\subsection{Multiplication operators via lifted operator-valued Clenshaw on the $D$-simplex}

\paragraph{Polynomial spaces.}
Let $D\ge 1$ and let $\kappa\in\mathbb{R}^{D+1}$ with $\kappa_r>-1/2$.
Let $\{\phi_\alpha^{(\kappa)}\}_{|\alpha|\le n}$ denote the orthonormal Jacobi basis on the simplex
$\Delta^D$, ordered in graded lexicographic order.
Define
\[
\Pi_n^D := \operatorname{span}\{\phi_\alpha^{(\kappa)} : |\alpha|\le n\},
\qquad
M_n := \dim \Pi_n^D .
\]
Let
\[
\mathcal H_j^D := \operatorname{span}\{\phi_\alpha^{(\kappa)} : |\alpha|=j\},
\qquad
m_j := \dim \mathcal H_j^D = \binom{j+D-1}{D-1}.
\]
Then
\[
\Pi_n^D = \bigoplus_{j=0}^n \mathcal H_j^D,
\qquad
M_n = \sum_{j=0}^n m_j .
\]

For each $j$, let $E_j\in\mathbb{R}^{M_n\times m_j}$ denote the canonical embedding of the
degree-$j$ block, and $E_j^T$ the corresponding extraction.

\paragraph{Coordinate Jacobi matrices.}
For each coordinate $x_i$, $i=1,\dots,D$, define the Jacobi (multiplication) matrix
\[
J_i^{(n)} \in \mathbb{R}^{M_n\times M_n},
\qquad
(J_i^{(n)})_{\ell k}
=
\langle \phi_\ell^{(\kappa)},\, x_i\,\phi_k^{(\kappa)}\rangle_{w_\kappa}.
\]
Since the basis is orthonormal and multiplication by $x_i$ is self-adjoint,
\[
J_i^{(n)} = (J_i^{(n)})^T .
\]
Moreover, multiplication by a coordinate changes total degree by at most one, so
\[
E_r^T J_i^{(n)} E_s = 0
\quad\text{unless}\quad r\in\{s-1,s,s+1\}.
\]

Define the degree blocks
\[
\begin{aligned}
	A_{i,j} &:= E_j^T J_i^{(p)} E_j \in \mathbb{R}^{m_j\times m_j},\\
	B_{i,j} &:= E_j^T J_i^{(p)} E_{j+1} \in \mathbb{R}^{m_j\times m_{j+1}},\\
	C_{i,j} &:= E_j^T J_i^{(p)} E_{j-1} = B_{i,j-1}^T .
\end{aligned}
\]

\paragraph{Multiplication as operator functional calculus.}
Let
\[
f(x) = \sum_{|\alpha|\le N} c_\alpha\,\phi_\alpha^{(\kappa)}(x),
\qquad
q(x) = \sum_{|\beta|\le p} q_\beta\,\phi_\beta^{(\kappa)}(x).
\]
Formally,
\[
q(x) f(x)
=
\Phi(x)^T\, q(J_1,\dots,J_D)\, c .
\]
To evaluate this stably and without aliasing, we work in a lifted space
$\Pi_K^D$ with
\[
K \ge N + p,
\]
embed $c$ into $\Pi_K^D$, apply the multiplication operator there, and then restrict back
to $\Pi_N^D$.

\paragraph{Lifted unknowns.}
Define lifted vectors
\[
v_j \in \mathbb{R}^{m_j M_K},
\qquad j=0,1,\dots,p,
\]
and stack
\[
v :=
\begin{bmatrix}
	v_0\\ v_1\\ \vdots\\ v_p
\end{bmatrix}
\in \mathbb{R}^{M_p M_K}.
\]
Each $v_j$ represents the action of $\Phi_j(J)$ on function-space coefficients in $\Pi_K^D$.

\paragraph{Stacked degree recurrence.}
For each $j\ge 0$, the coordinate recurrences
\[
x_i \Phi_j
=
\Phi_{j-1} C_{i,j}
+
\Phi_j A_{i,j}
+
\Phi_{j+1} B_{i,j},
\qquad i=1,\dots,D,
\]
are stacked into
\[
\mathcal C_j \Phi_{j-1}
+
\mathcal A_j \Phi_j
-
(I \otimes J^{(K)}) \Phi_j
+
\mathcal B_j \Phi_{j+1}
= 0,
\]
where
\[
\mathcal A_j :=
\begin{bmatrix}
	A_{1,j}\\ \vdots\\ A_{D,j}
\end{bmatrix},
\quad
\mathcal B_j :=
\begin{bmatrix}
	B_{1,j}\\ \vdots\\ B_{D,j}
\end{bmatrix},
\quad
\mathcal C_j :=
\begin{bmatrix}
	C_{1,j}\\ \vdots\\ C_{D,j}
\end{bmatrix}.
\]
This system has $D m_j$ equations but only $m_{j+1}$ unknowns in $\Phi_{j+1}$.

\paragraph{Canonical pivot rule.}
Let $\alpha=(\alpha_1,\dots,\alpha_D)$ denote the graded-lex multi-index.
From the $D m_j$ stacked equations, select exactly $m_{j+1}$ rows by:
\begin{itemize}
	\item taking all rows from coordinate $i=1$,
	\item from coordinate $i=2$, taking only those basis indices with $\alpha_1=0$,
	\item from coordinate $i=3$, those with $\alpha_1=\alpha_2=0$,
	\item $\vdots$
	\item from coordinate $i=D$, those with $\alpha_1=\cdots=\alpha_{D-1}=0$.
\end{itemize}
The total number of selected rows is
\[
m_j + \binom{j+D-2}{D-2} + \cdots + 1
=
\binom{j+D}{D-1}
=
m_{j+1}.
\]

Let $P_j$ denote the row-selection operator implementing this pivot rule.
Define the effective blocks
\[
A_j := P_j \mathcal A_j,
\qquad
B_j := P_j \mathcal B_j,
\qquad
C_j := P_j \mathcal C_j .
\]

\paragraph{Lifted block system.}
Define the block operator $L_{p,K}(J)\in\mathbb{R}^{(M_p M_K)\times(M_p M_K)}$ by:
\begin{itemize}
	\item the $(0,0)$ block is $I_{M_K}$;
	\item for $j=0,\dots,p-1$, the $(j+1)$-st block row is
	\[
	C_j \otimes I_{M_K}
	\;+\;
	A_j \otimes I_{M_K}
	\;-\;
	(I \otimes J^{(K)})
	\;+\;
	B_j \otimes I_{M_K},
	\]
	acting on $(v_{j-1},v_j,v_{j+1})$.
\end{itemize}
This operator is square and block lower triangular.

\paragraph{Central identity (operator-valued Clenshaw).}
Let $q\in\mathbb{R}^{M_p}$ and let $c^{(K)}\in\mathbb{R}^{M_K}$ be the embedding of
$c\in\mathbb{R}^{M_N}$ into $\Pi_K^D$.
Then
\[
\boxed{
	\begin{aligned}
		L_{p,K}(J)^T v &= q \otimes c^{(K)},\\
		M_q^{(K)} c^{(K)} &= (e_0^T \otimes I)\, v .
\end{aligned}}
\]
Finally, the de-aliased multiplication operator on $\Pi_N^D$ is obtained by restriction:
\[
\boxed{
	M_q^{(N)} c
	=
	R_{K\to N}\,
	\bigl(M_q^{(K)} E_{N\to K} c\bigr).
}
\]

\paragraph{Interpretation.}
This construction evaluates the multiplication operator $q(J)$ by a lifted,
block-triangular solve (operator-valued Clenshaw), avoiding aliasing by performing
all intermediate operations in $\Pi_K^D$ before restriction.

\subsubsection{Block back substitution (operator Clenshaw) without assembling $L_{p,K}(J)$}

We keep the notation of the previous section.  Fix $D\ge 1$, degrees $p\ge 0$ and $K\ge 0$,
and let $M_p=\dim\Pi_p^D$, $M_K=\dim\Pi_K^D$, $m_j=\dim\mathcal H_j^D$.

\paragraph{Lifted unknowns and right-hand side.}
For each $j=0,1,\dots,p$ define
\[
v_j \in \mathbb{R}^{m_j M_K},
\qquad
v := [v_0;v_1;\dots;v_p] \in \mathbb{R}^{M_p M_K}.
\]
Let the coefficient blocks of $q\in\Pi_p^D$ be $q_j\in\mathbb{R}^{m_j}$ so that
$q=[q_0;\dots;q_p]\in\mathbb{R}^{M_p}$.
Let $c^{(K)}\in\mathbb{R}^{M_K}$ denote the embedding of $c\in\mathbb{R}^{M_N}$ into $\Pi_K^D$
(zeros in degrees $>N$).  Define the block right-hand side
\[
r_j := q_j \otimes c^{(K)} \in \mathbb{R}^{m_j M_K},
\qquad
r := [r_0;\dots;r_p] = q\otimes c^{(K)} \in \mathbb{R}^{M_p M_K}.
\]

\paragraph{Row selection (canonical pivot rule).}
For each $j=0,\dots,p-1$ we stack the $D$ coordinate equations, yielding $D m_j$ rows.
Let $P_j$ be the row-selection operator that picks exactly $m_{j+1}$ rows (the pivot rule used in code):
\begin{itemize}
	\item from coordinate block $i=1$: all $m_j$ rows,
	\item from coordinate block $i=2$: only those with $\alpha_1=0$,
	\item from coordinate block $i=3$: only those with $\alpha_1=\alpha_2=0$,
	\item $\vdots$
	\item from coordinate block $i=D$: only those with $\alpha_1=\cdots=\alpha_{D-1}=0$,
\end{itemize}
so that the total number of selected rows is $m_{j+1}$.

Equivalently, there exists a deterministic map
\[
\sigma_j:\{1,\dots,m_{j+1}\}\to\{1,\dots,D\}\times\{1,\dots,m_j\},
\qquad
\sigma_j(\ell)=(i_\ell,b_\ell),
\]
such that the $\ell$-th selected row comes from coordinate $i_\ell$ and within-degree index $b_\ell$.

\paragraph{Degree blocks from Jacobi matrices.}
Using the Jacobi matrices in $\Pi_p^D$, define the per-coordinate blocks
\[
A_{i,j} := E_j^T J_i^{(p)} E_j \in \mathbb{R}^{m_j\times m_j},\qquad
B_{i,j} := E_j^T J_i^{(p)} E_{j+1} \in \mathbb{R}^{m_j\times m_{j+1}},\qquad
C_{i,j} := B_{i,j-1}^T .
\]
Stack across coordinates:
\[
\mathcal A_j := \begin{bmatrix} A_{1,j}\\ \vdots\\ A_{D,j}\end{bmatrix}\in\mathbb{R}^{(D m_j)\times m_j},\quad
\mathcal B_j := \begin{bmatrix} B_{1,j}\\ \vdots\\ B_{D,j}\end{bmatrix}\in\mathbb{R}^{(D m_j)\times m_{j+1}},\quad
\mathcal C_j := \begin{bmatrix} C_{1,j}\\ \vdots\\ C_{D,j}\end{bmatrix}\in\mathbb{R}^{(D m_j)\times m_{j-1}}.
\]
Apply the pivot rule:
\[
A_j := P_j \mathcal A_j \in \mathbb{R}^{m_{j+1}\times m_j},\qquad
B_j := P_j \mathcal B_j \in \mathbb{R}^{m_{j+1}\times m_{j+1}},\qquad
C_j := P_j \mathcal C_j \in \mathbb{R}^{m_{j+1}\times m_{j-1}} \ (j\ge 1).
\]

\paragraph{The row-picked function-space multiplication operator.}
Let $J_i^{(K)}\in\mathbb{R}^{M_K\times M_K}$ be the Jacobi matrices in $\Pi_K^D$
(the function-space multiplication operators).  Define a linear operator
\[
\mathcal J_j : \mathbb{R}^{m_j M_K}\to \mathbb{R}^{m_{j+1} M_K}
\]
by the canonical row map $\sigma_j$:
write any $u\in\mathbb{R}^{m_j M_K}$ in $m_j$ blocks $u=[u^{(1)};\dots;u^{(m_j)}]$ with $u^{(b)}\in\mathbb{R}^{M_K}$,
and define
\[
(\mathcal J_j u)^{(\ell)} := J_{i_\ell}^{(K)}\, u^{(b_\ell)} \in \mathbb{R}^{M_K},
\qquad \ell=1,\dots,m_{j+1},
\]
where $\sigma_j(\ell)=(i_\ell,b_\ell)$.
This is exactly the action assembled in the code by placing the row-picked blocks $-(I\otimes J_i^{(K)})$
into the middle block column of $L_{p,K}(J)$.

The adjoint $\mathcal J_j^T:\mathbb{R}^{m_{j+1}M_K}\to\mathbb{R}^{m_jM_K}$ satisfies
\[
(\mathcal J_j^T w)^{(b)} \;=\;\sum_{\ell:\,b_\ell=b} (J_{i_\ell}^{(K)})^T\, w^{(\ell)}
\;=\;\sum_{\ell:\,b_\ell=b} J_{i_\ell}^{(K)}\, w^{(\ell)},
\]
using symmetry of $J_i^{(K)}$.

\paragraph{Implicit block operator $L_{p,K}(J)$.}
Define the block-lower-triangular operator $L_{p,K}(J)$ (never explicitly assembled for the back-substitution form)
by its action on block vectors $[v_0;\dots;v_p]$:
\begin{align*}
	\text{block row }0:\qquad & I_{M_K}\, v_0,\\
	\text{block row }(j+1),\ j=0,\dots,p-1:\qquad
	& (C_j\otimes I_{M_K})\, v_{j-1}
	\;+\;
	\bigl(A_j\otimes I_{M_K} - \mathcal J_j\bigr)\, v_j
	\;+\;
	(B_j\otimes I_{M_K})\, v_{j+1}.
\end{align*}
(For $j=0$, the $C_0$ term is absent.)

\paragraph{Central solve and extraction.}
We compute $v$ from the \emph{upper-triangular} system
\[
L_{p,K}(J)^T v = r = q\otimes c^{(K)},
\]
and return
\[
M_q^{(K)} c^{(K)} = (e_0^T\otimes I)\, v = v_0 \in \mathbb{R}^{M_K}.
\]
Finally, the de-aliased multiplication operator on $\Pi_N^D$ is
\[
M_q^{(N)}c = R_{K\to N}\, v_0 .
\]

\paragraph{Block back substitution (operator Clenshaw).}
Because $L_{p,K}(J)$ is block lower triangular with diagonal blocks
\[
D_0 := I_{M_K},\qquad
D_j := B_{j-1}\otimes I_{M_K}\ \ (j=1,\dots,p),
\]
the transposed system $L_{p,K}(J)^T v=r$ is block upper triangular.
Introduce the conventions $v_{p+1}\equiv 0$ and $v_{p+2}\equiv 0$.
Then the back substitution proceeds for $j=p,p-1,\dots,0$ as follows:

\begin{itemize}
	\item For $j=p$:
	\[
	(B_{p-1}^T\otimes I_{M_K})\, v_p = r_p.
	\]
	\item For $1\le j\le p-1$:
	\[
	(B_{j-1}^T\otimes I_{M_K})\, v_j
	=
	r_j
	-
	\bigl(A_j\otimes I_{M_K} - \mathcal J_j\bigr)^T v_{j+1}
	-
	(C_{j+1}^T\otimes I_{M_K})\, v_{j+2}.
	\]
	\item For $j=0$:
	\[
	v_0
	=
	r_0
	-
	\bigl(A_0\otimes I_{M_K} - \mathcal J_0\bigr)^T v_1
	-
	(C_{1}^T\otimes I_{M_K})\, v_{2}.
	\]
\end{itemize}

\paragraph{Diagonal solves are small (tensor structure).}
Since $D_j=B_{j-1}\otimes I_{M_K}$, solving
\[
(B_{j-1}^T\otimes I_{M_K}) v_j = s_j
\]
reduces to applying $B_{j-1}^{-T}$ on the polynomial-block index while leaving the
$\Pi_K^D$ (function-space) dimension untouched:
if $v_j=[v_j^{(1)};\dots;v_j^{(m_j)}]$ and $s_j=[s_j^{(1)};\dots;s_j^{(m_j)}]$ with blocks in $\mathbb{R}^{M_K}$,
then for each fixed function-space coordinate $t\in\{1,\dots,M_K\}$,
\[
\begin{bmatrix}
	v_j^{(1)}[t]\\ \vdots\\ v_j^{(m_j)}[t]
\end{bmatrix}
=
B_{j-1}^{-T}
\begin{bmatrix}
	s_j^{(1)}[t]\\ \vdots\\ s_j^{(m_j)}[t]
\end{bmatrix}.
\]
In practice, $B_{j-1}\in\mathbb{R}^{m_j\times m_j}$ is small and dense, so we precompute
a dense factorization of $B_{j-1}^T$ and apply it to each of the $M_K$ right-hand sides.

\paragraph{Remarks.}
\begin{itemize}
	\item This recurrence is exactly the block back substitution for the system solved in code by
	\texttt{spsolve(L.T, $q \otimes c$)}; it avoids assembling $L_{p,K}(J)$ by applying only the
	block operators $A_j,B_j,C_j$ and $\mathcal J_j,\mathcal J_j^T$.
	\item The only places the large $M_K\times M_K$ Jacobi matrices appear are in the applications
	of $\mathcal J_j$ and $\mathcal J_j^T$, which are implemented by sparse SpMV with the $J_i^{(K)}$.
	\item \color{red} We actually find that $B_{j-1}$ are sparse - lower triangular and banded with a main diagonal, and only non-zeros below the diagonal in the last few row blocks. Hence, we implement a simple back substitution routine on the sparse upper-triangular system. \color{black}
\end{itemize}
